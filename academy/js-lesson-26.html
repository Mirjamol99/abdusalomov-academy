Function parametr va argumentlari

Function faqat o‘zining ichidagi ma'lumotlar bilan ishlamaydi.

Biz functionga tashqaridan ma'lumot berishimiz mumkin.

Buning uchun parametr va argument tushunchalaridan foydalanamiz.

Parametr nima? 🤔

Function yaratilayotganda qavs ichida yoziladigan o‘zgaruvchi parametr deyiladi. 

function salom(ism) {

    console.log("Salom, " + ism);

}

Bu yerda ism — parametr.

U functionga keyinchalik beriladigan qiymatni qabul qiladi.

Argument nima? 📥

Function chaqirilganda qavs ichida beriladigan haqiqiy qiymat argument deyiladi. 

function salom(ism) {

    console.log("Salom, " + ism);

}

salom("Mirjamol");

Bu yerda:

ism → parametr
"Mirjamol" → argument

Parametr va argument farqi ⚖️

📦 Parametr — function yaratilganda yoziladi.


📥 Argument — function chaqirilganda beriladi. 

function salom(ism) {
    //      ↑
    //   parametr

    console.log(ism);
}

salom("Ali");
//     ↑
//  argument

Bitta parametr 🔹 

function yoshniChiqar(yosh) {

    console.log("Yosh:", yosh);

}

yoshniChiqar(16);
Yosh: 16

Ikkita parametr 🔹🔹 

function tanishtir(ism, yosh) {

    console.log("Ism:", ism);
    console.log("Yosh:", yosh);

}

tanishtir("Ali", 16);
Ism: Ali
Yosh: 16

Uchta parametr 🔹🔹🔹 

function userInfo(ism, yosh, shahar) {

    console.log("Ism:", ism);
    console.log("Yosh:", yosh);
    console.log("Shahar:", shahar);

}

userInfo("Ali", 16, "Qarshi");

Parametrlar tartibi muhim ⚠️

Argumentlar parametrlar tartibida joylashadi. 

function info(ism, yosh) {

    console.log(ism);
    console.log(yosh);

}

info("Ali", 15);

Birinchi argument birinchi parametrga, ikkinchi argument esa ikkinchi parametrga tushadi.

Noto‘g‘ri tartib bersak nima bo‘ladi? 🤔 

function info(ism, yosh) {

    console.log("Ism:", ism);
    console.log("Yosh:", yosh);

}

info(16, "Ali");
Ism: 16
Yosh: Ali

JavaScript argumentlarni avtomatik ravishda "to‘g‘rilab" bermaydi.

Shuning uchun parametrlar tartibiga e'tibor berish kerak.

Sonlar bilan ishlash 🔢 

function qoshish(a, b) {

    console.log(a + b);

}

qoshish(10, 20);
30

a va b — parametrlar.
10 va 20 — argumentlar.

Ayirish, ko‘paytirish va bo‘lish ➗ 

function ayirish(a, b) {
    console.log(a - b);
}

function kopaytirish(a, b) {
    console.log(a * b);
}

function bolish(a, b) {
    console.log(a / b);
}

ayirish(20, 5);
kopaytirish(4, 5);
bolish(20, 4);
15
20
5

Argument sifatida o‘zgaruvchi 📦

Argument sifatida faqat to‘g‘ridan-to‘g‘ri qiymat emas, o‘zgaruvchini ham berish mumkin. 

let ism = "Ali";
let yosh = 16;

function info(ism, yosh) {

    console.log(ism);
    console.log(yosh);

}

info(ism, yosh);

Expressionni argument sifatida berish 🧠 

function natija(son) {

    console.log(son);

}

natija(10 + 5);
15

JavaScript avval 10 + 5 ni hisoblaydi, keyin natijani functionga beradi.

Function natijasini argument qilish 🔗 

function qoshish(a, b) {

    return a + b;

}

function kopaytirish(a, b) {

    return a * b;

}

let natija = kopaytirish(
    qoshish(2, 3),
    4
);

console.log(natija);
20

Avval qoshish(2, 3) ishlaydi va 5 qaytaradi.

Keyin kopaytirish(5, 4) ishlaydi.

Default parameter ⚙️

Agar argument berilmasa, parametrga oldindan qiymat berishimiz mumkin. 

function salom(ism = "Mehmon") {

    console.log("Salom, " + ism);

}

salom();
salom("Ali");
Salom, Mehmon
Salom, Ali

Bir parametr berib, boshqasini bermasak 📥 

function info(ism, yosh) {

    console.log(ism);
    console.log(yosh);

}

info("Ali");
Ali
undefined

yosh uchun argument berilmagani sababli uning qiymati undefined bo‘ladi.

Ko‘p argument berish ⚠️ 

function salom(ism) {

    console.log("Salom, " + ism);

}

salom("Ali", 16, "Qarshi");

Bu holatda functionda faqat ism parametri bor.

Ortiqcha argumentlar oddiy parametrlar bilan bog‘lanmaydi.

Parametr sifatida Array 📦 

function arrayniChiqar(arr) {

    for (let item of arr) {

        console.log(item);

    }

}

let mevalar = [
    "Olma",
    "Banan",
    "Uzum"
];

arrayniChiqar(mevalar);

Parametr sifatida Object 🧩 

function userInfo(user) {

    console.log(user.name);
    console.log(user.age);

}

let user = {
    name: "Ali",
    age: 16
};

userInfo(user);
Ali
16

Boolean parametr bilan ishlash ✅ 

function loginTekshir(isLogin) {

    if (isLogin) {

        console.log("Xush kelibsiz!");

    } else {

        console.log("Iltimos, tizimga kiring.");

    }

loginTekshir(true);
loginTekshir(false);

Parametr va return 🔄

Parametr orqali ma'lumotni function ichiga olib kiramiz.

return orqali natijani functiondan tashqariga chiqaramiz. 

function kvadrat(son) {

    return son * son;

}

let natija = kvadrat(6);

console.log(natija);
36

Bir xil functionni turli qiymatlar bilan ishlatish 🔥 

function kvadrat(son) {

    return son * son;

}

console.log(kvadrat(2));
console.log(kvadrat(5));
console.log(kvadrat(10));
console.log(kvadrat(20));
4
25
100
400

Functionning eng katta afzalliklaridan biri — bir xil kodni qayta yozmasdan, turli qiymatlar bilan ishlatishdir.

Amaliy loyiha: Juft yoki toq 🔢

        return "Juft son";

}

Parametrlar bilan ishlash qoidalari 📌

1️⃣ Parametr function yaratilganda qavs ichida yoziladi.

2️⃣ Argument function chaqirilganda beriladi.

3️⃣ Argumentlar parametrlar tartibida joylashadi.

4️⃣ Argument berilmasa, parametr qiymati undefined bo‘lishi mumkin.

5️⃣ Default parameter yordamida parametrga boshlang‘ich qiymat berish mumkin.

6️⃣ Parametr sifatida String, Number, Boolean, Array, Object va boshqa qiymatlarni berish mumkin.

7️⃣ Function natijasini return orqali qaytarish mumkin.

Uyga vazifa 📝

1-vazifa:
Ism qabul qiladigan salom() function yarating.

2-vazifa:
Ism va yosh qabul qiladigan function yarating.

3-vazifa:
Uchta son qabul qilib, ularning yig‘indisini return qiling.

4-vazifa:
Son qabul qilib, uning kvadratini qaytaring.

5-vazifa:
Sonning juft yoki toq ekanini aniqlang.

6-vazifa:
Array qabul qiladigan function yarating va barcha elementlarni Console'ga chiqaring.

7-vazifa:
Object qabul qiladigan function yarating va uning name, age qiymatlarini chiqaring.

8-vazifa:
Default parameter ishlatib, argument berilmaganda "Mehmon" chiqaring.

9-vazifa:
Ikki son qabul qiladigan kalkulyator functionlari yarating.

10-vazifa:
Ism, yosh va ball qabul qiladigan student function yarating va ma'lumotlarni chiroyli qilib Console'ga chiqaring.

Bugungi dars xulosasi 🎯

📥 Parametr — function qabul qiladigan o‘zgaruvchi.

🎯 Argument — functionga beriladigan haqiqiy qiymat.

🔢 Bir nechta parametr ishlatish mumkin.

📦 Argument sifatida o‘zgaruvchi, Array yoki Object berish mumkin.

⚙️ Default parameter argument berilmaganda boshlang‘ich qiymat beradi.

↩️ return orqali natijani qaytarish mumkin.
