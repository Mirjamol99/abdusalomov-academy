📦 1. Array yaratamiz

Bizning barcha vazifalarimiz bitta Array ichida saqlanadi. 

let tasks = [];

📦 2. Task Object

Har bir vazifa Object bo‘ladi. 

let task = {

    id: Date.now(),

    text: "JavaScript o‘rganish",

    completed: false

};

➕ 3. push()

Yangi vazifani Array'ga push() orqali qo‘shamiz. 

tasks.push(task);

💾 4. LocalStorage

Vazifalar sahifa yangilanganda yo‘qolmasligi uchun LocalStorage'ga saqlaymiz. 

localStorage.setItem(
    "tasks",
    JSON.stringify(tasks)
);

📤 5. Ma'lumotni olish 

let tasks =
    JSON.parse(
        localStorage.getItem("tasks")
    ) || [];
🔥 || [] juda muhim.

Agar LocalStorage'da ma'lumot bo‘lmasa, bo‘sh Array ishlatiladi.

🎨 6. Vazifalarni ekranga chiqarish

Array ichidagi vazifalarni HTML'ga chiqarish uchun forEach() ishlatamiz. 

tasks.forEach(function(task) {

    console.log(task.text);

});

🔎 7. filter()

Faqat bajarilgan yoki bajarilmagan vazifalarni olish uchun filter() ishlatamiz. 

let completed =
    tasks.filter(function(task) {

        return task.completed === true;

    });

🗑️ 8. Vazifani o‘chirish

Kerakli vazifani filter() orqali olib tashlaymiz. 

tasks =
    tasks.filter(function(task) {

        return task.id !== id;

    });

✅ 9. Vazifani bajarilgan qilish 

task.completed =
    !task.completed;

! boolean qiymatni teskarisiga o‘zgartiradi.

⚡ 10. To‘liq JavaScript 

let tasks =
    JSON.parse(
        localStorage.getItem("tasks")
    ) || [];


let currentFilter = "all";


let todoForm =
    document.querySelector("#todoForm");


let todoInput =
    document.querySelector("#todoInput");


let taskList =
    document.querySelector("#taskList");


let taskCount =
    document.querySelector("#taskCount");


let clearBtn =
    document.querySelector("#clearBtn");



function saveTasks() {

    localStorage.setItem(
        "tasks",
        JSON.stringify(tasks)
    );

}



function renderTasks() {

    taskList.innerHTML = "";


    let filteredTasks = tasks;


    if (currentFilter === "active") {

        filteredTasks =
            tasks.filter(function(task) {

                return task.completed === false;

            });

    }


    if (currentFilter === "completed") {

        filteredTasks =
            tasks.filter(function(task) {

                return task.completed === true;

            });

    }


    if (filteredTasks.length === 0) {

        taskList.innerHTML = `
            <div
                class="text-center
                       text-slate-500
                       py-10">

                📭 Vazifalar topilmadi.

            </div>
        `;

    }


    filteredTasks.forEach(
        function(task) {


            let div =
                document.createElement("div");


            div.className = `
                task-card
                bg-slate-950
                border
                border-slate-800
                rounded-2xl
                p-5
                flex
                items-center
                gap-4
            `;


            div.innerHTML = `

                <button
                    class="
                        complete-btn
                        w-10
                        h-10
                        rounded-full
                        border
                        border-slate-600
                        flex
                        items-center
                        justify-center
                    "
                    data-id="${task.id}"
                >

                    ${task.completed ? "✓" : ""}

                </button>


                <p
                    class="
                        task-text
                        flex-1
                        font-semibold
                        ${task.completed
                            ? "task-done"
                            : ""}
                    "
                >

                    ${task.text}

                </p>


                <button
                    class="
                        delete-btn
                        text-red-400
                        hover:text-red-300
                        px-3
                        py-2
                    "
                    data-id="${task.id}"
                >

                    🗑️

                </button>

            `;


            taskList.appendChild(div);

        }
    );


    let activeCount =
        tasks.filter(function(task) {

            return task.completed === false;

        }).length;


    taskCount.textContent =
        activeCount +
        " ta bajarilmagan vazifa";

}



/* =====================================
   ADD TASK
===================================== */

todoForm.addEventListener(
    "submit",
    function(event) {

        event.preventDefault();


        let text =
            todoInput.value.trim();


        if (text === "") {

            alert(
                "Vazifa yozing!"
            );

            return;

        }


        let newTask = {

            id: Date.now(),

            text: text,

            completed: false

        };


        tasks.push(newTask);


        saveTasks();


        todoInput.value = "";


        renderTasks();

    }
);



/* =====================================
   COMPLETE + DELETE
===================================== */

taskList.addEventListener(
    "click",
    function(event) {


        let completeButton =
            event.target.closest(
                ".complete-btn"
            );


        let deleteButton =
            event.target.closest(
                ".delete-btn"
            );



        if (completeButton) {

            let id =
                Number(
                    completeButton.dataset.id
                );


            tasks =
                tasks.map(
                    function(task) {


                        if (
                            task.id === id
                        ) {

                            return {
                                ...task,

                                completed:
                                    !task.completed
                            };

                        }


                        return task;

                    }
                );


            saveTasks();


            renderTasks();

        }



        if (deleteButton) {

            let id =
                Number(
                    deleteButton.dataset.id
                );


            tasks =
                tasks.filter(
                    function(task) {

                        return task.id !== id;

                    }
                );


            saveTasks();


            renderTasks();

        }

    }
);



/* =====================================
   FILTERS
===================================== */

let filterButtons =
    document.querySelectorAll(
        ".filter-btn"
    );


filterButtons.forEach(
    function(button) {

        button.addEventListener(
            "click",
            function() {


                currentFilter =
                    button.dataset.filter;


                filterButtons.forEach(
                    function(btn) {

                        btn.classList.remove(
                            "bg-yellow-400",
                            "text-slate-950"
                        );

                        btn.classList.add(
                            "bg-slate-800",
                            "text-slate-300"
                        );

                    }
                );


                button.classList.remove(
                    "bg-slate-800",
                    "text-slate-300"
                );


                button.classList.add(
                    "bg-yellow-400",
                    "text-slate-950"
                );


                renderTasks();

            }
        );

    }
);



/* =====================================
   CLEAR ALL
===================================== */

clearBtn.addEventListener(
    "click",
    function() {


        if (tasks.length === 0) {

            return;

        }


        let answer =
            confirm(
                "Barcha vazifalar o‘chirilsinmi?"
            );


        if (!answer) {

            return;

        }


        tasks = [];


        saveTasks();


        renderTasks();

    }
);



/* =====================================
   FIRST RENDER
===================================== */

renderTasks();
