return

return — function ichidan qiymatni tashqariga qaytarish uchun ishlatiladi.

Oddiy qilib aytganda, function biror hisob-kitobni bajaradi va return yordamida natijani bizga qaytaradi.

return nima? 🤔

Function ichida return ishlatilsa, undan keyingi qiymat functiondan tashqariga qaytariladi. 

function qoshish(a, b) {

    return a + b;

}

let natija = qoshish(10, 20);

console.log(natija);
30

return qanday ishlaydi? ⚙️

1️⃣ Function chaqiriladi.


2️⃣ Function hisob-kitobni bajaradi.


3️⃣ return natijani qaytaradi.


4️⃣ Qaytgan qiymatni o‘zgaruvchiga saqlash yoki boshqa joyda ishlatish mumkin.

console.log va return farqi ⚖️

Bu ikkalasini adashtirmaslik juda muhim. 

function test1() {

    console.log(10);

}

function test2() {

    return 20;

}

test1();

let result = test2();

console.log(result);

console.log() qiymatni Console'ga chiqaradi.

return esa qiymatni functiondan tashqariga qaytaradi.

return qiymatini o‘zgaruvchiga saqlash 📦 

function qoshish(a, b) {

    return a + b;

}

let result = qoshish(5, 7);

console.log(result);
12

Bu yerda functiondan qaytgan 12 qiymati result o‘zgaruvchisiga saqlandi.

return bilan matematik amallar ➕ 

function ayirish(a, b) {

    return a - b;

}

function kopaytirish(a, b) {

    return a * b;

}

function bolish(a, b) {

    return a / b;

}

console.log(ayirish(20, 5));
console.log(kopaytirish(4, 5));
console.log(bolish(20, 4));
15
20
5

return bilan String 📝 

function salom(ism) {

    return "Salom, " + ism;

}

let result = salom("Mirjamol");

console.log(result);
Salom, Mirjamol

return bilan Template Literal 📝 

function tanishtir(ism, yosh) {

    return `Mening ismim ${ism}.
Men ${yosh} yoshdaman.`;

}

console.log(
    tanishtir("Ali", 16)
);

return Boolean qiymat qaytarishi ✅ 

function juftmi(son) {

    return son % 2 === 0;

}

console.log(juftmi(10));
console.log(juftmi(7));
true
false

Function true yoki false qaytarishi mumkin.

return va if 🔀 

function baho(score) {

    if (score >= 90) {

        return "A";

    }

    if (score >= 80) {

        return "B";

    }

    if (score >= 70) {

        return "C";

    }

    return "F";

}

console.log(baho(95));
console.log(baho(82));
console.log(baho(65));
A
B
F

return functionni to‘xtatadi 🛑

return bajarilgandan keyin functionning qolgan qismi ishlamaydi. 

function test() {

    console.log("Birinchi");

    return;

    console.log("Ikkinchi");

}

test();
Birinchi

"Ikkinchi" Console'ga chiqmaydi, chunki undan oldin return bajarildi.

return dan keyin kod yozish ⚠️ 

function test() {

    return 100;

    console.log("Bu ishlamaydi");

}

Odatda returndan keyingi kod function ichida bajarilmaydi.

return va expression 🧠 

function hisobla(a, b) {

    return a * b + 10;

}

console.log(
    hisobla(5, 4)
);
30

return faqat bitta oddiy qiymat emas, expression natijasini ham qaytarishi mumkin.

return va boshqa function 🔗 

function qoshish(a, b) {

    return a + b;

}

function kvadrat(son) {

    return son * son;

}

let result = kvadrat(
    qoshish(2, 3)
);

console.log(result);
25

Avval:
qoshish(2, 3) → 5

Keyin:
kvadrat(5) → 25

function mevalar() {

    return [
        "Olma",
        "Banan",
        "Uzum"
    ];

let fruits = mevalar();

console.log(fruits);

return va Object 🧩 

function user yarat() {

    return {
        name: "Ali",
        age: 16
    };

}

Function Object ham qaytarishi mumkin.

Masalan: 

function userInfo() {

    return {
        name: "Ali",
        age: 16
    };

}

let user = userInfo();

console.log(user.name);
console.log(user.age);

return va undefined ❓

Agar function ichida return bo‘lmasa, function odatda undefined qaytaradi. 

function test() {

    console.log("Salom");

}

let result = test();

console.log(result);
Salom
undefined

Bo‘sh return ↩️ 

function tekshir(yosh) {

    if (yosh < 18) {

        return;

    }

    console.log("Kirish mumkin");

}

tekshir(15);

return; qiymat qaytarmaydi, lekin functionni shu joyda to‘xtatadi.

Erta return — Early Return 🚀

Ba'zan shart bajarilmasa, functionni darhol to‘xtatish juda qulay. 

function login(isLogin) {

    if (!isLogin) {

        return "Avval login qiling";

    }

    return "Xush kelibsiz!";

}

console.log(login(false));
console.log(login(true));
Avval login qiling
Xush kelibsiz!

return bilan tekshirish 🔍 

function kattaSon(a, b) {

    if (a > b) {

        return a;

    }

    return b;

}

console.log(kattaSon(20, 15));
20

return bilan yoshni tekshirish 👤 

function yoshTekshir(yosh) {

    if (yosh >= 18) {

        return "Voyaga yetgan";

    }

    return "Voyaga yetmagan";

}

console.log(yoshTekshir(20));
console.log(yoshTekshir(15));

function juftmi(son) {

        : "Toq";

console.log(juftmi(7));

Amaliy loyiha: Kalkulyator 🧮 

function qoshish(a, b) {
    return a + b;
}

    return a - b;

Eng muhim qoida 📌

return — functiondan qiymat qaytaradi va shu bilan function bajarilishini to‘xtatadi. 

function misol() {

    return "Salom";

    // Bu kod ishlamaydi
    console.log("Hello");

}

Uyga vazifa 📝

1-vazifa:
Ikkita son qabul qilib, ularning yig‘indisini return qiladigan function yarating.

2-vazifa:
Sonning kvadratini return qiling.

3-vazifa:
Son juft bo‘lsa true, toq bo‘lsa false qaytaradigan function yarating.

4-vazifa:
Ikki sondan kattasini return qiladigan function yarating.

5-vazifa:
Ism qabul qilib, Salom, [ism]! ko‘rinishida qiymat qaytaring.

6-vazifa:
Array qabul qilib, uning birinchi elementini return qiling.

7-vazifa:
Yosh qabul qiladigan function yarating. Agar yosh 18 yoki undan katta bo‘lsa "Kirish mumkin", aks holda "Kirish mumkin emas" qaytarsin.

8-vazifa:
Uchta sonning o‘rtacha qiymatini return qiladigan function yarating.

9-vazifa:
Login holatini tekshiradigan function yarating va true/falsega qarab turli qiymat qaytaring.

10-vazifa:
Kalkulyator yarating: +, -, *, /. Har bir amal natijasini return qilsin.
