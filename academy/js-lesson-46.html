Object property va methodlari

Bugungi darsda Object ichidagi property va method tushunchalarini chuqurroq o‘rganamiz.

Shuningdek, Object.keys(), Object.values() va Object.entries() metodlari bilan tanishamiz.

📦 Property nima?

Object ichida saqlanadigan key-value ma'lumot property deyiladi. 

let user = {

    name: "Ali",

    age: 18,

    city: "Andijon"

};
Bu Object'da 3 ta property bor:

name
age
city

🔍 Property'ni olish

Property qiymatini nuqta . orqali olish mumkin. 

let user = {

    name: "Ali",

    age: 18

};

console.log(user.name);

console.log(user.age);
Ali
18

🧩 Bracket notation

Property'ni [] orqali ham olish mumkin. 

let user = {

    name: "Ali",

    age: 18

};

console.log(user["name"]);

console.log(user["age"]);
Quyidagi ikkala yozuv ham bir xil:

user.name
user["name"]

✏️ Property'ni o‘zgartirish 

let user = {

    name: "Ali",

    age: 18

};

user.age = 20;

console.log(user.age);
20

➕ Property qo‘shish 

let user = {

    name: "Ali",

    age: 18

};

user.city = "Andijon";

console.log(user);
{
    name: "Ali",
    age: 18,
    city: "Andijon"
}

❌ Property o‘chirish 

let user = {

    name: "Ali",

    age: 18,

    city: "Andijon"

};

delete user.city;

console.log(user);

⚙️ Method nima?

Object ichida saqlangan function method deyiladi.

Ya'ni:

property → ma'lumot
method → amal bajaruvchi function 

let user = {

    name: "Ali",

    sayHello: function() {

        console.log("Salom!");

    }

};

user.sayHello();
Salom!

⚡ Methodni qisqa yozish

Object methodini zamonaviy JavaScript'da qisqa ko‘rinishda yozish mumkin. 

let user = {

    name: "Ali",

    sayHello() {

        console.log("Salom!");

    }

};

user.sayHello();
Bu:

sayHello: function() {}

bilan bir xil ma'noga ega.

👤 Method va this

Object methodi ichida this orqali shu Object'ning property'lariga murojaat qilish mumkin. 

let user = {

    name: "Ali",

    age: 18,

    sayHello() {

        console.log(
            "Salom, " + this.name
        );

    }

};

user.sayHello();
Salom, Ali
this.name → user.name

🧠 this bilan age 

let user = {

    name: "Ali",

    age: 18,

    info() {

        console.log(
            "Ism: " + this.name
        );

        console.log(
            "Yosh: " + this.age
        );

    }

};

user.info();
Ism: Ali
Yosh: 18

🔑 Object.keys()

Object.keys() Object ichidagi barcha key'larni Array ko‘rinishida qaytaradi. 

let user = {

    name: "Ali",

    age: 18,

    city: "Andijon"

};

let keys = Object.keys(user);

console.log(keys);
[
    "name",
    "age",
    "city"
]

💎 Object.values()

Object.values() Object ichidagi barcha value'larni Array ko‘rinishida qaytaradi. 

let user = {

    name: "Ali",

    age: 18,

    city: "Andijon"

};

let values = Object.values(user);

console.log(values);
[
    "Ali",
    18,
    "Andijon"
]

🔗 Object.entries()

Object.entries() key va value'larni juftlik ko‘rinishida Array qilib beradi. 

let user = {

    name: "Ali",

    age: 18

};

console.log(
    Object.entries(user)
);
[
    ["name", "Ali"],
    ["age", 18]
]

🔄 Object.keys() bilan loop 

let user = {

    name: "Ali",

    age: 18,

    city: "Andijon"

};

let keys = Object.keys(user);

keys.forEach(function(key) {

    console.log(
        key,
        user[key]
    );

});
name Ali
age 18
city Andijon

🔁 Object.values() bilan loop 

let user = {

    name: "Ali",

    age: 18,

    city: "Andijon"

};

Object.values(user).forEach(
    function(value) {

        console.log(value);

    }
);

🔗 Object.entries() bilan loop 

let user = {

    name: "Ali",

    age: 18,

    city: "Andijon"

};

Object.entries(user).forEach(
    function(entry) {

        console.log(
            entry[0],
            entry[1]
        );

    }
);
entry[0] → key

entry[1] → value

🛒 Product Object 

let product = {

    name: "Pizza",

    price: 50000,

    category: "Pizza",

    available: true,

    showInfo() {

        console.log(
            "Mahsulot: " +
            this.name
        );

        console.log(
            "Narx: " +
            this.price
        );

    }

};

product.showInfo();
Mahsulot: Pizza
Narx: 50000

👤 User Object methodi 

let user = {

    name: "Ali",

    age: 18,

    isOnline: true,

    login() {

        this.isOnline = true;

        console.log(
            this.name +
            " tizimga kirdi."
        );

    },

    logout() {

        this.isOnline = false;

        console.log(
            this.name +
            " tizimdan chiqdi."
        );

    }

};

user.login();

user.logout();

🧩 Property va Method farqi

Property

Ma'lumot saqlaydi. 

let user = {

    name: "Ali",

    age: 18

};

Method

Amal bajaradi. 

let user = {

    sayHello() {

        console.log(
            "Salom!"
        );

    }

};

🔥 To‘liq amaliy misol 

let student = {

    name: "Ali",

    age: 18,

    score: 95,

    isStudent: true,

    subjects: [
        "Math",
        "English",
        "JavaScript"
    ],

    getInfo() {

        console.log(
            "Ism: " + this.name
        );

        console.log(
            "Yosh: " + this.age
        );

        console.log(
            "Ball: " + this.score
        );

    },

    isPassed() {

        return this.score >= 60;

    }

};


student.getInfo();

console.log(
    student.isPassed()
);
Ism: Ali
Yosh: 18
Ball: 95

true

Object.entries(object)

Key va value'larni juftlik qilib beradi.

🎯 Dars xulosasi

📦 property → Object ichidagi ma'lumot.

⚙️ method → Object ichidagi function.

🔍 Object.keys() → key'larni qaytaradi.

💎 Object.values() → value'larni qaytaradi.

🔗 Object.entries() → key va value juftliklarini qaytaradi.

🧠 this → method ichida Object'ning o‘ziga murojaat qilishga yordam beradi.
