Eventlar va addEventListener

Bugun foydalanuvchi sahifada biror harakat qilganda JavaScript qanday javob berishini o‘rganamiz.

Asosiy mavzular: click, addEventListener(), input, change, submit, keydown va event obyekti.

🖱️ Event nima?

Event — foydalanuvchining sahifadagi harakati yoki brauzerda sodir bo‘ladigan hodisa.

🖱️ click

Element ustiga bosilganda ishlaydi.

⌨️ keydown

Klaviaturadagi tugma bosilganda ishlaydi.

📝 input

Input qiymati o‘zgarganda ishlaydi.

📋 submit

Form yuborilganda ishlaydi.

⚡ addEventListener()

addEventListener() elementga event qo‘shish uchun ishlatiladi. 

element.addEventListener(
    "click",
    function () {

        console.log("Bosildi!");

    }
);
Bu yerda:

"click" → event turi.

function () { } → event sodir bo‘lganda bajariladigan kod.

🖱️ click event

Eng ko‘p ishlatiladigan eventlardan biri click hisoblanadi. 

let button =
    document.querySelector("#btn");

button.addEventListener(
    "click",
    function () {

        console.log("Tugma bosildi!");

    }
);

🔥 Amaliy misol — click

Hali bosilmadi. 

let button =
    document.querySelector("#clickBtn");

let result =
    document.querySelector("#clickResult");


button.addEventListener(
    "click",
    function () {

        result.textContent =
            "🎉 Tugma bosildi!";

    }
);

📝 Event bilan function

Event sodir bo‘lganda alohida functionni ham chaqirish mumkin. 

function sayHello() {

    console.log("Salom!");

}


let button =
    document.querySelector("#btn");


button.addEventListener(
    "click",
    sayHello
);
⚠️ Bu yerda:

To‘g‘ri: sayHello

Noto‘g‘ri: sayHello()

🖱️ mouseover

Sichqoncha element ustiga kelganda mouseover ishlaydi. 

box.addEventListener(
    "mouseover",
    function () {

        console.log(
            "Sichqoncha keldi!"
        );

    }
);
mouseover → sichqoncha element ustiga kelganda.

🖱️ mouseout

Sichqoncha elementdan chiqib ketganda mouseout ishlaydi. 

box.addEventListener(
    "mouseout",
    function () {

        console.log(
            "Sichqoncha chiqdi!"
        );

    }
);

⌨️ keydown

Klaviaturadagi tugma bosilganda keydown ishlaydi. 

document.addEventListener(
    "keydown",
    function () {

        console.log(
            "Klaviatura tugmasi bosildi!"
        );

    }
);
💡 Bu event butun sahifada klaviatura bilan ishlash uchun ham qo‘llanadi.

🔤 keydown va event.key

Qaysi tugma bosilganini event.key orqali bilish mumkin. 

document.addEventListener(
    "keydown",
    function (event) {

        console.log(event.key);

    }
);

Masalan, foydalanuvchi A tugmasini bossa: 

A

📝 input event

Input ichidagi qiymat o‘zgargan sari input event ishlaydi. 

let input =
    document.querySelector("#name");

input.addEventListener(
    "input",
    function (event) {

        console.log(
            event.target.value
        );

    }
);

🔥 Amaliy misol — input

Ismingiz shu yerda chiqadi. 

let input =
    document.querySelector("#nameInput");

let result =
    document.querySelector("#nameResult");


input.addEventListener(
    "input",
    function (event) {

        result.textContent =
            event.target.value;

    }
);

📋 change event

change odatda input yoki select qiymati o‘zgarganda ishlaydi. 

let select =
    document.querySelector("#city");

select.addEventListener(
    "change",
    function (event) {

        console.log(
            event.target.value
        );

    }
);

📋 Amaliy misol — select

Shahar tanlanmagan. 

let select =
    document.querySelector("#city");

let result =
    document.querySelector("#cityResult");


select.addEventListener(
    "change",
    function (event) {

        result.textContent =
            "Tanlangan shahar: " +
            event.target.value;

    }
);

📤 submit event

Form yuborilganda submit event ishlaydi. 

let form =
    document.querySelector("#form");


form.addEventListener(
    "submit",
    function (event) {

        event.preventDefault();

        console.log(
            "Form yuborildi!"
        );

    }
);
event.preventDefault() brauzerning formani odatdagi yuborish harakatini to‘xtatadi.

🎯 Event object

Event sodir bo‘lganda JavaScript bizga event haqida ma'lumot beruvchi object beradi. 

button.addEventListener(
    "click",
    function (event) {

        console.log(event);

    }
);

🎯 event.target

event.target event sodir bo‘lgan elementni bildiradi. 

button.addEventListener(
    "click",
    function (event) {

        console.log(
            event.target
        );

    }
);
💡 Masalan, qaysi button bosilganini event.target orqali olish mumkin.

🔥 Amaliy misol — event.target

Button tanlang. 

let buttons =
    document.querySelectorAll(
        ".targetBtn"
    );


buttons.forEach(function(button) {

    button.addEventListener(
        "click",
        function (event) {

            document.querySelector(
                "#targetResult"
            ).textContent =
                event.target.textContent;

        }
    );

});

🚫 removeEventListener()

Event listenerni olib tashlash uchun removeEventListener() ishlatiladi. 

function hello() {

    console.log("Salom!");

}


button.addEventListener(
    "click",
    hello
);


button.removeEventListener(
    "click",
    hello
);
⚠️ Olib tashlash uchun bir xil function reference ishlatilishi kerak.
